cmake_minimum_required(VERSION 3.20)
project(TCCCompilerPOC LANGUAGES C)
set(POC_ATOMIC_BACKEND "auto" CACHE STRING "Atomic32 backend: auto, gnu, gas-x86_64 or win32")
set_property(CACHE POC_ATOMIC_BACKEND PROPERTY STRINGS auto gnu gas-x86_64 win32)
if(POC_ATOMIC_BACKEND STREQUAL "auto")
  if(WIN32)
    set(POC_ATOMIC_BACKEND "win32")
  else()
    set(POC_ATOMIC_BACKEND "gnu")
  endif()
endif()
if(POC_ATOMIC_BACKEND STREQUAL "win32")
  if(NOT WIN32)
    message(FATAL_ERROR "win32 backend requires Windows")
  endif()
  set(POC_ATOMIC_IMPL src/atomic/win_interlocked.c)
elseif(POC_ATOMIC_BACKEND STREQUAL "gas-x86_64")
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
    message(FATAL_ERROR "gas-x86_64 backend requires Linux x86-64")
  endif()
  enable_language(ASM)
  set(POC_ATOMIC_IMPL src/atomic/x86_64_sysv.S)
elseif(POC_ATOMIC_BACKEND STREQUAL "gnu")
  if(NOT CMAKE_C_COMPILER_ID MATCHES "^(GNU|Clang)$")
    message(FATAL_ERROR "GNU builtin backend requires GCC/Clang; TCC cannot fall back")
  endif()
  set(POC_ATOMIC_IMPL src/atomic/gnu_atomic.c)
else()
  message(FATAL_ERROR "Unsupported backend: ${POC_ATOMIC_BACKEND}")
endif()
add_library(poc_atomic32 STATIC ${POC_ATOMIC_IMPL})
target_include_directories(poc_atomic32 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
include(CTest)
if(BUILD_TESTING)
  add_executable(atomic_thread_test tests/atomic_thread.c)
  target_link_libraries(atomic_thread_test PRIVATE poc_atomic32)
  if(NOT WIN32)
    find_package(Threads REQUIRED)
    target_link_libraries(atomic_thread_test PRIVATE Threads::Threads)
  endif()
  add_test(NAME atomic_thread COMMAND atomic_thread_test)
  set_tests_properties(atomic_thread PROPERTIES TIMEOUT 60)
  if(WIN32)
    add_executable(win_api_test tests/win_api.c)
    add_test(NAME win_api COMMAND win_api_test)
    add_library(poc_bridge SHARED tests/bridge.c)
    target_include_directories(poc_bridge PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
    add_executable(bridge_client tests/bridge_client.c)
    target_include_directories(bridge_client PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
    add_test(NAME bridge_client COMMAND bridge_client)
  endif()
endif()
message(STATUS "TCC compiler POC atomic backend: ${POC_ATOMIC_BACKEND}")
