cmake_minimum_required(VERSION 3.20)
project(TCCCompilerPOC LANGUAGES C)
set(POC_ATOMIC_BACKEND "auto" CACHE STRING "Atomic32 backend: auto, gnu, gas-x86_64, gas-aarch64 or win32")
set_property(CACHE POC_ATOMIC_BACKEND PROPERTY STRINGS auto gnu gas-x86_64 gas-aarch64 win32)
if(POC_ATOMIC_BACKEND STREQUAL "auto")
  if(WIN32)
    set(POC_ATOMIC_BACKEND "win32")
  else()
    set(POC_ATOMIC_BACKEND "gnu")
  endif()
endif()
if(POC_ATOMIC_BACKEND STREQUAL "win32")
  if(NOT WIN32)
    message(FATAL_ERROR "win32 backend requires Windows")
  endif()
  set(POC_ATOMIC_IMPL src/atomic/win_interlocked.c)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "64-bit pointer/atomic POC does not support Win32 x86")
  endif()
  enable_language(ASM)
  set(POC_ATOMIC64_IMPL src/atomic/win_x64.S)
  set(POC_ATOMIC_SMALL_IMPL src/atomic/gnu_atomic_small.c)
elseif(POC_ATOMIC_BACKEND STREQUAL "gas-x86_64")
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
    message(FATAL_ERROR "gas-x86_64 backend requires Linux x86-64")
  endif()
  enable_language(ASM)
  set(POC_ATOMIC_IMPL src/atomic/x86_64_sysv.S)
  set(POC_ATOMIC64_IMPL src/atomic/x86_64_sysv64.S)
  set(POC_ATOMIC_SMALL_IMPL src/atomic/x86_64_sysv_small.S)
elseif(POC_ATOMIC_BACKEND STREQUAL "gas-aarch64")
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
    message(FATAL_ERROR "gas-aarch64 backend requires native Linux AArch64")
  endif()
  enable_language(ASM)
  set(POC_ATOMIC_IMPL src/atomic/aarch64_linux.S)
  set(POC_ATOMIC64_IMPL src/atomic/aarch64_linux64.S)
  set(POC_ATOMIC_SMALL_IMPL src/atomic/aarch64_linux_small.S)
elseif(POC_ATOMIC_BACKEND STREQUAL "gnu")
  if(NOT CMAKE_C_COMPILER_ID MATCHES "^(GNU|Clang)$")
    message(FATAL_ERROR "GNU builtin backend requires GCC/Clang; TCC cannot fall back")
  endif()
  set(POC_ATOMIC_IMPL src/atomic/gnu_atomic.c)
  set(POC_ATOMIC64_IMPL src/atomic/gnu_atomic64.c)
  set(POC_ATOMIC_SMALL_IMPL src/atomic/gnu_atomic_small.c)
else()
  message(FATAL_ERROR "Unsupported backend: ${POC_ATOMIC_BACKEND}")
endif()
add_library(poc_threads STATIC src/threads/threads.c)
target_include_directories(poc_threads PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(NOT WIN32)
  find_package(Threads REQUIRED)
  target_link_libraries(poc_threads PUBLIC Threads::Threads)
endif()
add_library(poc_atomic_small STATIC ${POC_ATOMIC_SMALL_IMPL})
target_include_directories(poc_atomic_small PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_library(poc_atomic64 STATIC ${POC_ATOMIC64_IMPL} src/atomic/ptr64.c)
target_include_directories(poc_atomic64 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_library(poc_atomic32 STATIC ${POC_ATOMIC_IMPL})
target_include_directories(poc_atomic32 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
include(CTest)
if(BUILD_TESTING)
  add_executable(atomic_small_thread_test tests/atomic_small_thread.c)
  target_link_libraries(atomic_small_thread_test PRIVATE poc_atomic_small poc_threads)
  add_test(NAME atomic_small_thread COMMAND atomic_small_thread_test)
  set_tests_properties(atomic_small_thread PROPERTIES TIMEOUT 60)
  add_executable(atomic64_thread_test tests/atomic64_thread.c)
  target_link_libraries(atomic64_thread_test PRIVATE poc_atomic64 poc_threads)
  add_test(NAME atomic64_thread COMMAND atomic64_thread_test)
  set_tests_properties(atomic64_thread PROPERTIES TIMEOUT 60)
  add_executable(thread_shutdown_overlap_test tests/thread_shutdown_overlap.c)
  target_link_libraries(thread_shutdown_overlap_test PRIVATE poc_threads)
  add_test(NAME thread_shutdown_overlap COMMAND thread_shutdown_overlap_test)
  set_tests_properties(thread_shutdown_overlap PROPERTIES TIMEOUT 60)
  add_executable(thread_contract_test tests/thread_contract.c)
  target_link_libraries(thread_contract_test PRIVATE poc_threads poc_atomic32)
  add_test(NAME thread_contract COMMAND thread_contract_test)
  set_tests_properties(thread_contract PROPERTIES TIMEOUT 60)
  add_executable(atomic_thread_test tests/atomic_thread.c)
  target_link_libraries(atomic_thread_test PRIVATE poc_atomic32)
  if(NOT WIN32)
    find_package(Threads REQUIRED)
    target_link_libraries(atomic_thread_test PRIVATE Threads::Threads)
  endif()
  add_test(NAME atomic_thread COMMAND atomic_thread_test)
  set_tests_properties(atomic_thread PROPERTIES TIMEOUT 60)
  if(WIN32)
    add_executable(win_api_test tests/win_api.c)
    add_test(NAME win_api COMMAND win_api_test)
    add_library(poc_bridge SHARED tests/bridge.c)
    set_target_properties(poc_bridge PROPERTIES PREFIX "")
    target_include_directories(poc_bridge PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
    add_executable(bridge_client tests/bridge_client.c)
    target_include_directories(bridge_client PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
    add_test(NAME bridge_client COMMAND bridge_client)
  endif()
endif()
message(STATUS "TCC compiler POC atomic backend: ${POC_ATOMIC_BACKEND}")
